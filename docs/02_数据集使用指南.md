D1 = fs.getNode("D1").mat()  # 畸变

"""
统一注册数据集加载器
支持: BOP LM/T-LESS, SCARED, EndoNeRF, Stereo_Lap, 棋盘格GT, BlendedMVS, DTU
"""
from dataclasses import dataclass
from pathlib import Path
import json, numpy as np, cv2, open3d as o3d

@dataclass
class RegSample:
    """注册样本: 图像 + (可选深度) + (可选内参) + (可选GT位姿) + (可选3D模型)"""
    image: np.ndarray              # HxWx3 RGB
    depth: np.ndarray = None       # HxW float32 (米)
    K: np.ndarray = None           # 3x3内参
    pose_gt: np.ndarray = None     # 4x4 T_obj^cam
    mesh: o3d.geometry.TriangleMesh = None
    mask: np.ndarray = None        # HxW uint8


class BOPLMLoader:
    """BOP Linemod 加载器"""
    ROOT = Path(r"D:\bop\lm")

    def __init__(self, scene_id=1):
        self.scene_dir = self.ROOT / "test" / f"{scene_id:06d}"
        with open(self.scene_dir / "scene_camera.json") as f:
            self.cams = json.load(f)
        with open(self.scene_dir / "scene_gt.json") as f:
            self.gts = json.load(f)

    def __len__(self): return len(self.cams)

    def __getitem__(self, idx) -> RegSample:
        s = str(idx)
        img = cv2.imread(str(self.scene_dir / "rgb" / f"{s:06d}.png"))
        img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
        depth = cv2.imread(str(self.scene_dir / "depth" / f"{s:06d}.png"),
                           cv2.IMREAD_UNCHANGED).astype(np.float32)
        depth /= self.cams[s]["depth_scale"]  # 转米
        K = np.array(self.cams[s]["cam_K"]).reshape(3,3)
        # GT位姿(第一个物体)
        g = self.gts[s][0]
        R = np.array(g["cam_R_m2c"]).reshape(3,3)
        t = np.array(g["cam_t_m2c"]).reshape(3,1)
        pose = np.eye(4); pose[:3,:3] = R; pose[:3,3] = t.flatten()
        # 加载模型
        mesh = o3d.io.read_triangle_mesh(
            str(self.ROOT / "models" / f"obj_{g['obj_id']:06d}.ply"))
        return RegSample(image=img, depth=depth, K=K, pose_gt=pose, mesh=mesh)


class SCAREDLoader:
    """SCARED 内窥镜RGB-D加载器"""
    def __init__(self, dataset_id=1, keyframe_id=1,
                 root=r"E:\MIS_Datasets\SCARED"):
        self.kf = Path(root) / f"dataset_{dataset_id}" / f"keyframe_{keyframe_id}"
        fs = cv2.FileStorage(str(self.kf / "endoscope_calibration.yaml"),
                             cv2.FILE_STORAGE_READ)
        self.M1 = fs.getNode("M1").mat()
        self.D1 = fs.getNode("D1").mat()

    def load(self) -> RegSample:
        img = cv2.imread(str(self.kf / "Left_Image.png"))
        img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
        depth = cv2.imread(str(self.kf / "left_depth_map.tiff"),
                           cv2.IMREAD_UNCHANGED).astype(np.float32) / 1000.0
        mesh = o3d.io.read_triangle_mesh(str(self.kf / "point_cloud.obj"))
        return RegSample(image=img, depth=depth, K=self.M1, mesh=mesh)


class ChessboardGTLoader:
    """棋盘格GT位姿加载器"""
    ROOT = Path(r"E:\MIS_Datasets\chessboard")

    def __getitem__(self, idx) -> RegSample:
        img = cv2.imread(str(self.ROOT / "images" / f"photo_{idx+1:03d}.jpg"))
        img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
        pose = np.loadtxt(self.ROOT / "gt_poses" / f"pose_{idx:04d}.txt")
        return RegSample(image=img, pose_gt=pose)


class VGGTReconLoader:
    """VGGT重建结果加载器(无标定入口)"""
    ROOT = Path(r"E:\MIS_TMI_Re_3D\vggt_chessboard")

    def __init__(self):
        self.K = np.loadtxt(self.ROOT / "vggt_intrinsics.txt")

    def __getitem__(self, idx) -> RegSample:
        img = cv2.imread(str(self.ROOT / "vggt_depths" / f"depth_{idx:04d}.png"),
                         cv2.IMREAD_UNCHANGED)
        depth = img.astype(np.float32) / 1000.0
        pose = np.loadtxt(self.ROOT / "vggt_poses" / f"c2w_{idx:04d}.txt")
        return RegSample(depth=depth, K=self.K, pose_gt=pose)

### 11.3 SLAM位姿加载（视频序列）

```python
"""ORB-SLAM3轨迹输出格式 → 注册管线"""
import numpy as np
from pathlib import Path

def load_orbslam_trajectory(traj_file):
    """
    ORB-SLAM3轨迹格式: timestamp tx ty tz qx qy qz qw
    Returns: list of (t, T_cam^world)
    """
    poses = []
    with open(traj_file) as f:
        for line in f:
            parts = line.strip().split()
            if len(parts) != 8: continue
            t = float(parts[0])
            tx,ty,tz = map(float, parts[1:4])
            qx,qy,qz,qw = map(float, parts[4:8])
            # 四元数 → 旋转矩阵
            R = quat_to_rot(qx,qy,qz,qw)
            T = np.eye(4); T[:3,:3]=R; T[:3,3]=[tx,ty,tz]
            poses.append((t, T))
    return poses

def quat_to_rot(qx,qy,qz,qw):
    """四元数转3x3旋转矩阵"""
    n = np.sqrt(qx*qx+qy*qy+qz*qz+qw*qw)
    qx,qy,qz,qw = qx/n,qy/n,qz/n,qw/n
    return np.array([
        [1-2*(qy*qy+qz*qz), 2*(qx*qy-qz*qw),   2*(qx*qz+qy*qw)],
        [2*(qx*qy+qz*qw),   1-2*(qx*qx+qz*qz), 2*(qy*qz-qx*qw)],
        [2*(qx*qz-qy*qw),   2*(qy*qz+qx*qw),   1-2*(qx*qx+qy*qy)]
    ])
```
