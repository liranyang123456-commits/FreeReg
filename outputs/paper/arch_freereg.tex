% 3D block, 以 (cx, cy) 为几何中心, 宽 w 高 h, 标签, 主色
\newcommand{\blockc}[7]{%
  % #1 name, #2 cx, #3 cy, #4 w, #5 h, #6 label, #7 front color
  \begin{scope}[shift={(#2,#3)}]
    \def\dx{0.28}\def\dy{0.28}
    \pgfmathsetmacro{\w}{#4}\pgfmathsetmacro{\h}{#5}
    \fill[blkSide] (\w/2,-\h/2) -- (\w/2+\dx,-\h/2+\dy) -- (\w/2+\dx,\h/2+\dy) -- (\w/2,\h/2) -- cycle;
    \fill[blkTop] (-\w/2,\h/2) -- (-\w/2+\dx,\h/2+\dy) -- (\w/2+\dx,\h/2+\dy) -- (\w/2,\h/2) -- cycle;
    \fill[#7] (-\w/2,-\h/2) rectangle (\w/2,\h/2);
    \draw[blkEdge, line width=0.4pt] (-\w/2,-\h/2) rectangle (\w/2,\h/2);
    \node[align=center, font=\footnotesize] at (0,0) {#6};
    \coordinate (#1-west) at (-\w/2,0);
    \coordinate (#1-east) at (\w/2+\dx,0);
    \coordinate (#1-south) at (0,-\h/2);
    \coordinate (#1-north) at (0,\h/2);
  \end{scope}%
}
